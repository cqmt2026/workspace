```python
import jqdatasdk as jq
```

### 3.1 回测框架

```python
# backtest_framework.py
# 回测框架

class BacktestFramework:
    """回测框架"""
    
    def __init__(self, initial_capital=100000):
        self.initial_capital = initial_capital
        self.capital = initial_capital
        self.positions = {}
        self.trades = []
        self.daily_returns = []
    
    def run_backtest(self, strategy, stock_list, start_date, end_date):
        """运行回测"""
        print(f"开始回测：{start_date} - {end_date}")
        print(f"初始资金：{self.initial_capital}")
        print(f"策略：{strategy.name}")
        
        # 1. 获取历史数据
        data = self.get_historical_data(stock_list, start_date, end_date)
        
        # 2. 每日执行策略
        for date in data['dates']:
            # 生成交易信号
            signals = strategy.generate_signals(data, date)
            
            # 执行交易
            self.execute_trades(signals, date, data)
            
            # 更新持仓
            self.update_positions(date, data)
            
            # 计算当日收益
            daily_return = self.calculate_daily_return(date)
            self.daily_returns.append(daily_return)
        
        # 3. 计算绩效指标
        performance = self.calculate_performance()
        
        return performance
    
    def calculate_performance(self):
        """计算绩效指标"""
        # 总收益率
        total_return = (self.capital - self.initial_capital) / self.initial_capital
        
        # 年化收益率
        trading_days = len(self.daily_returns)
        annual_return = (1 + total_return) ** (252 / trading_days) - 1
        
        # 夏普比率
        if np.std(self.daily_returns) > 0:
            sharpe_ratio = np.mean(self.daily_returns) / np.std(self.daily_returns) * np.sqrt(252)
        else:
            sharpe_ratio = 0
        
        # 最大回撤
        max_drawdown = self.calculate_max_drawdown()
        
        # 胜率
        win_rate = self.calculate_win_rate()
        
        # 盈亏比
        profit_loss_ratio = self.calculate_profit_loss_ratio()
        
        return {
            'total_return': total_return,
            'annual_return': annual_return,
            'sharpe_ratio': sharpe_ratio,
            'max_drawdown': max_drawdown,
            'win_rate': win_rate,
            'profit_loss_ratio': profit_loss_ratio,
            'trading_days': trading_days,
            'total_trades': len(self.trades)
        }
    
    def calculate_max_drawdown(self):
        """计算最大回撤"""
        cumulative_returns = np.cumprod(1 + np.array(self.daily_returns))
        running_max = np.maximum.accumulate(cumulative_returns)
        drawdown = (cumulative_returns - running_max) / running_max
        return np.min(drawdown)
    
    def calculate_win_rate(self):
        """计算胜率"""
        if len(self.trades) == 0:
            return 0
        
        winning_trades = sum(1 for trade in self.trades if trade['profit'] > 0)
        return winning_trades / len(self.trades)
    
    def calculate_profit_loss_ratio(self):
        """计算盈亏比"""
        winning_profits = [trade['profit'] for trade in self.trades if trade['profit'] > 0]
        losing_profits = [abs(trade['profit']) for trade in self.trades if trade['profit'] < 0]
        
        if len(losing_profits) == 0:
            return 0
        
        avg_win = np.mean(winning_profits) if winning_profits else 0
        avg_loss = np.mean(losing_profits) if losing_profits else 1
        
        return avg_win / avg_loss
```

```python
# strategy_backtest.py
# 策略回测模块

class StrategyBacktest:
    """策略回测模块"""
    
    def __init__(self):
        self.strategies = {}
    
    def register_strategy(self, name, strategy):
        """注册策略"""
        self.strategies[name] = strategy
    
    def backtest_single_strategy(self, strategy_name, start_date, end_date, initial_capital=100000):
        """回测单个策略"""
        if strategy_name not in self.strategies:
            print(f"策略{strategy_name}不存在")
            return None
        
        strategy = self.strategies[strategy_name]
        backtest = BacktestFramework(initial_capital=initial_capital)
        
        stock_list = self.get_strategy_stock_list(strategy_name)
        performance = backtest.run_backtest(strategy, stock_list, start_date, end_date)
        
        return performance
    
    def compare_strategies(self, strategy_names, start_date, end_date):
        """对比多个策略"""
        results = {}
        
        for strategy_name in strategy_names:
            print(f"\n回测策略：{strategy_name}")
            performance = self.backtest_single_strategy(strategy_name, start_date, end_date)
            results[strategy_name] = performance
        
        # 生成对比报告
        self.generate_comparison_report(results)
        
        return results
    
    def optimize_parameters(self, strategy_name, param_ranges, start_date, end_date):
        """参数优化"""
        if strategy_name not in self.strategies:
            return None
        
        best_performance = None
        best_params = None
        
        # 网格搜索最优参数
        for params in self.generate_param_combinations(param_ranges):
            self.strategies[strategy_name].set_parameters(params)
            performance = self.backtest_single_strategy(strategy_name, start_date, end_date)
            
            if best_performance is None or performance['sharpe_ratio'] > best_performance['sharpe_ratio']:
                best_performance = performance
                best_params = params
        
        return best_params, best_performance
```
