# Tushare 数据接口

import tushare as ts

class TushareData:
    """Tushare 数据接口"""
    
    def __init__(self, token=None):
        self.token = token
        self.api = None
        
        if token:
            ts.set_token(token)
            self.api = ts.pro_api()
            print("✅ Tushare 初始化成功")
        else:
            print("⚠️ 未配置 Tushare token，使用基础接口")
            self.api = None
    
    def get_stock_list(self):
        """获取股票列表"""
        try:
            if self.api:
                df = self.api.stock_basic(exchange='', list_status='L')
            else:
                # 使用基础接口
                df = ts.get_stock_basics()
            
            print(f"✅ 获取 Tushare 股票列表成功：{len(df)}只")
            return df
        except Exception as e:
            print(f"❌ 获取 Tushare 股票列表失败：{e}")
            return None
    
    def get_daily_data(self, ts_code, start_date, end_date):
        """获取日线数据"""
        try:
            if self.api:
                df = self.api.daily(ts_code=ts_code, start_date=start_date, end_date=end_date)
            else:
                df = ts.get_k_data(ts_code, start=start_date, end=end_date)
            
            print(f"✅ 获取{ts_code}日线数据成功：{len(df)}条")
            return df
        except Exception as e:
            print(f"❌ 获取{ts_code}日线数据失败：{e}")
            return None
    
    def get_financial_data(self, ts_code, start_date, end_date):
        """获取财务数据"""
        try:
            if self.api:
                df = self.api.fina_indicator(ts_code=ts_code, start_date=start_date, end_date=end_date)
            else:
                print("⚠️ 基础接口不支持财务数据")
                return None
            
            print(f"✅ 获取{ts_code}财务数据成功")
            return df
        except Exception as e:
            print(f"❌ 获取{ts_code}财务数据失败：{e}")
            return None

# 股票池定期清理模块

class StockPoolCleanup:
    """股票池清理"""
    
    def __init__(self):
        self.cleanup_rules = {
            'continuous_loss': 3,  # 连续亏损次数
            'technical_breakdown': True,  # 技术破位
            'liquidity_issue': True,  # 流动性问题
            'fundamental_issue': True  # 基本面问题
        }
    
    def cleanup_pool(self, pool_type='observation'):
        """清理股票池"""
        print(f"\n{'='*60}")
        print(f"股票池清理 ({pool_type})")
        print(f"{'='*60}")
        
        # 1. 加载股票池
        print("\n[步骤 1] 加载股票池...")
        stock_pool = self.load_stock_pool(pool_type)
        
        # 2. 检查清理规则
        print("\n[步骤 2] 检查清理规则...")
        stocks_to_remove = []
        
        for stock in stock_pool:
            if self.should_remove(stock):
                stocks_to_remove.append(stock)
        
        # 3. 执行清理
        print("\n[步骤 3] 执行清理...")
        for stock in stocks_to_remove:
            self.remove_stock(stock, pool_type)
            print(f"  移除：{stock['code']} {stock['name']}")
        
        # 4. 保存结果
        print("\n[步骤 4] 保存结果...")
        self.save_stock_pool(pool_type)
        
        print(f"\n{'='*60}")
        print(f"清理完成")
        print(f"移除股票：{len(stocks_to_remove)}只")
        print(f"{'='*60}")
        
        return stocks_to_remove
    
    def should_remove(self, stock):
        """判断是否应该移除"""
        # 连续亏损
        if stock.get('continuous_loss', 0) >= self.cleanup_rules['continuous_loss']:

        # 流动性问题

            return True
        
        # 基本面问题
        if self.cleanup_rules['fundamental_issue'] and stock.get('fundamental_issue', False):
