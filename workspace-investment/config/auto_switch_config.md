## 🤖 自动识别规则

### 【规则 1: 股票代码识别】

```python
import re

def detect_stock_code(text):
    """检测股票代码"""
    patterns = [
        r'\d{6}\.SZ',  # 深市
        r'\d{6}\.SH',  # 沪市
        r'\d{5}\.HK',  # 港股
        r'[A-Z]{4,5}', # 美股
    ]
    
    for pattern in patterns:
        match = re.search(pattern, text)
        if match:
            return match.group()
    return None
```

### 【规则 2: 关键词识别】

```python
def detect_investment_topic(text):
    """检测投资话题"""
    keywords_level1 = [
        '股票', '投资', '基金', 'ETF', '回测',
        '量化', '因子', '风控', '仓位', '止盈止损'
    ]
    
    keywords_level2 = [
        '涨跌', '买入', '卖出', '行情', '大盘',
        '板块', '龙头', '估值', '财报'
    ]
    
    text_lower = text.lower()
    
    # 一级关键词 (直接切入)
    for kw in keywords_level1:
        if kw in text_lower:
            return 'level1'
    
    # 二级关键词 (询问后切入)
    for kw in keywords_level2:
        if kw in text_lower:
            return 'level2'
    
    return None
```

### 【规则 3: 自动响应】

```python
def auto_respond_investment(text):
    """自动响应投资话题"""
    
    # 检测股票代码
    code = detect_stock_code(text)
    
    # 检测话题类型
    topic_level = detect_investment_topic(text)
    
    if code or topic_level == 'level1':
        # 直接切入策略工厂
        response = f"""
检测到投资相关话题！

📊 **投资分析策略工厂已就绪**

可用功能:
1. 30 因子量化评分
2. 专业风险评估
3. 智能信号生成
4. 5+2 策略系统
5. 专业回测验证

"""
        if code:
            response += f"检测到股票代码：{code}\n\n"
            response += "需要我进行以下分析吗？\n"
            response += "1. 因子评分\n"
            response += "2. 风险评估\n"
            response += "3. 信号生成\n"
            response += "4. 策略建议\n"
        
        return response
    
    elif topic_level == 'level2':
        # 询问是否需要策略工厂
        return """
检测到投资相关话题！

📊 **投资分析策略工厂可用**

需要我使用策略工厂进行分析吗？
- 30 因子量化评分
- 专业风险评估
- 智能信号生成
- 策略回测验证

回复"是"或具体需求即可开始分析。
"""
    
    return None
```
