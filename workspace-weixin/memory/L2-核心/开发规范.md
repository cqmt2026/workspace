## 一、文件组织规范

### 1. 页面文件结构
```
pages/
└── 模块名/
    ├── 模块名.js       ← 逻辑
    ├── 模块名.wxml     ← 结构
    ├── 模块名.wxss     ← 样式
    └── 模块名.json     ← 配置
```

### 2. 组件文件结构
```
components/
└── 组件名/
    ├── 组件名.js
    ├── 组件名.wxml
    ├── 组件名.wxss
    └── 组件名.json
```

### 3. 云函数文件结构
```
cloudfunctions/
└── 函数名/
    ├── index.js
    ├── package.json
    └── config.json
```

## 二、命名规范

### 1. 页面命名
- 全部小写，用横杠分隔
- ✅ `wrongbooks` / `add-question` / `practice-report`
- ❌ `WrongBooks` / `addQuestion` / `add_question`

### 2. 组件命名
- 组件名与文件名一致
- 组件js中：`Component({})`
- 注册标签时用 `kebab-case`

```xml
<!-- 使用组件 -->
<wrongbook-card id="card1"></wrongbook-card>
```

### 3. 云函数命名
- 全部小写，用横杠分隔
- ✅ `getWrongQuestions` / `addWrongQuestion`
- ❌ `GetWrongQuestions` / `add_wrong_question`

### 4. 数据库集合命名
- 全部小写
- ✅ `wrongbooks` / `practice_questions`
- ❌ `WrongBooks` / `practiceQuestions`

### 5. 变量命名
- JS变量：camelCase
- 常量：UPPER_SNAKE_CASE
- 私有变量：`_` 前缀

```javascript
// ✅ 正确示例
const cloudPath = 'ocr/' + Date.now() + '.jpg'
const MAX_RETRY = 3
let _timer = null

// ❌ 错误示例
const cloudpath = 'ocr/...'     // 不用 camelCase
const CloudPath = 'ocr/...'     // 不用 PascalCase
```

## 四、JS编码规范

### 1. 页面/组件结构（必须按顺序）
```javascript
// 1. 常量定义
const API_BASE = 'https://api.example.com'

// 2. Behavior引入
const myBehavior = require('./behaviors/myBehavior.js')

// 3. 页面的逻辑
Component({
  // 4. 开放属性
  properties: {
    myProp: { type: String, value: '' }
  },

  // 5. 组件私有数据
  data: { localData: null },

  // 6. 方法（按功能分组）
  methods: {
    // 6.1 事件处理
    onTap() { ... },
    onInput() { ... },
    
    // 6.2 业务逻辑
    fetchData() { ... },
    submit() { ... },
    
    // 6.3 工具方法
    formatDate() { ... }
  },

  // 7. 生命周期
  lifetimes: {
    created() { ... },
    attached() { ... },
    ready() { ... },
    detached() { ... }
  },

  // 8. Behavior
  behaviors: [myBehavior],

  // 9. 外部类（可选）
  externalClasses: ['my-class']
})
```

### 2. 云函数结构
```javascript
// 依赖声明
const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })

// 工具函数
function cleanText(text) { ... }

// 主入口（必须 exports.main）
exports.main = async function(event, context) {
  try {
    // 业务逻辑
    const result = await doSomething(event)
    return { success: true, data: result }
  } catch (err) {
    console.error('[ERROR]', err.message)
    return { success: false, errMsg: err.message }
  }
}
```

### 3. setData 规范
```javascript
// ✅ 正确：精确路径，不重复 setData
this.setData({ 'list[0].name': 'new' })

// ✅ 正确：合并多次 setData
const updates = {}
updates['form.question'] = 'xxx'
updates['form.answer'] = 'yyy'
this.setData(updates)

// ❌ 错误：setData 整个 data 对象
this.setData(this.data)

// ❌ 错误：频繁在循环内 setData
list.forEach(item => {
  this.setData({ [`item_${item.id}`]: process(item) })  // 错误
})
```

### 4. 异步操作规范
```javascript
// ✅ 正确：用 Promise 包装
uploadImage(filePath) {
  return new Promise((resolve, reject) => {
    wx.cloud.uploadFile({
      cloudPath, filePath,
      success: res => resolve(res),
      fail: reject
    })
  })
}

// ✅ 正确：用 async/await
async onConfirm() {
  try {
    wx.showLoading({ title: '处理中' })
    const croppedPath = await this.cropWithCanvas(...)
    const fileID = await this.uploadImage(croppedPath)
    const result = await this.doOCR(fileID)
    wx.hideLoading()
    this.setData({ ocrText: result.text })
  } catch (err) {
    wx.hideLoading()
    wx.showToast({ title: err.message, icon: 'none' })
  }
}
```

## 五、WXML 规范

### 1. 禁止在 WXML 中写业务逻辑
```xml
<!-- ❌ 错误：在 wxml 中计算 -->
<view wx:if="{{a + b > 10}}">...</view>

<!-- ✅ 正确：在 js 中计算好，传过来 -->
<view wx:if="{{isValid}}">...</view>
```

### 2. 循环必须加 key
```xml
<!-- ✅ 正确：加 key -->
<view wx:for="{{list}}" wx:for-item="item" wx:for-index="i" wx:key="id">
  {{item.name}}
</view>

<!-- ❌ 错误：无 key -->
<view wx:for="{{list}}">...</view>
```

### 3. 事件传参用 data-前缀
```xml
<!-- ✅ 正确：dataset 传参 -->
<view bindtap="onTap" data-id="{{item.id}}" data-name="{{item.name}}">
  点击
</view>

// js中
onTap(e) {
  const { id, name } = e.currentTarget.dataset
}
```

## 六、云开发规范

### 1. 云函数必须使用 DYNAMIC_CURRENT_ENV
```javascript
const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
```

### 2. 云存储路径规范
```
ocr/              ← OCR临时图片
uploads/          ← 用户上传图片
exports/          ← 导出文件
avatars/          ← 头像
```

### 3. 数据库权限原则
- 所有写操作通过云函数（云函数有admin权限）
- 客户端不直接操作数据库
- 集合权限：`onlyCreatorReadWrite`

## 七、错误处理规范

### 1. 必须 try-catch
```javascript
// ✅ 正确
try {
  const result = await wx.cloud.callFunction({ name: 'xxx', data: {...} })
  if (!result.result.success) throw new Error(result.result.errMsg)
} catch (err) {
  console.error('[ERROR]', err.message)
  wx.showToast({ title: err.message || '操作失败', icon: 'none' })
}

// ❌ 错误：无错误处理
wx.cloud.callFunction({ name: 'xxx', ... })
```

### 2. 统一的错误返回格式
```javascript
// 云函数统一返回格式
return { success: true, data: result }
return { success: false, errMsg: '错误原因' }
```

### 3. 用户提示
```javascript
// 加载提示
wx.showLoading({ title: '加载中...' })
wx.hideLoading()

// 错误提示
wx.showToast({ title: '操作失败', icon: 'none' })

// 成功提示
wx.showToast({ title: '保存成功' })
```

## 八、注释规范

### 1. 云函数必须注释参数
```javascript
/**
 * 添加错题
 * @param {string} question 题目内容
 * @param {string} answer   答案
 * @param {number} grade   年级（1-6）
 */
```

### 2. 复杂逻辑必须注释
```javascript
// 计算裁剪区域（屏幕坐标→实际像素坐标）
const scale = imgW / screenW
const cropX = Math.round(selLeft * scale)
```

### 3. 禁止无意义注释
```javascript
// ❌ 错误
// 这里是设置数据
this.setData({ value: 123 })

// ✅ 正确
// 标记已上传，后续不重复处理
this.setData({ uploaded: true })
```
